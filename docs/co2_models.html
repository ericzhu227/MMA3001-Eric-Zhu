co2_models

Compare indoor CO2 forecasts using chronological prepared datasets.

Models are selected by validation MAE only. The final test is scored after selection, without further tuning. All CO2 errors are in ppm. Timing is wall clock CPU time and varies with hardware. No scaling or imputation is learned from validation or test data.

 17def extract_environment_measurements(json_text, sensorid=None):
 18    """Extract selected measurements from one environmental JSON payload.
 19
 20    Args:
 21        json_text (str): JSON payload containing variable/value entries.
 22        sensorid (str, optional): Sensor identifier copied to each result.
 23
 24    Returns:
 25        list of tuple: ``(sensorid, epoch_seconds, variable, value)`` rows for
 26        CO2, temperature and humidity measurements.
 27
 28    Raises:
 29        ValueError: If the payload is not valid JSON.
 30        TypeError: If the payload is not a JSON list of measurement objects.
 31    """
 32    payload = json.loads(json_text)
 33    if not isinstance(payload, list):
 34        raise TypeError("Environmental JSON must contain a list.")
 35    selected = {"Carbon dioxide", "Temperature", "Humidity"}
 36    records = []
 37    for item in payload:
 38        if not isinstance(item, dict):
 39            raise TypeError("Each environmental entry must be an object.")
 40        variable = item.get("variable", {})
 41        if not isinstance(variable, dict):
 42            raise TypeError("Environmental variable metadata must be an object.")
 43        name = variable.get("name")
 44        if name not in selected:
 45            continue
 46        values = item.get("values", [])
 47        if not isinstance(values, list):
 48            raise TypeError("Environmental values must be a list.")
 49        for reading in values:
 50            if not isinstance(reading, dict):
 51                raise TypeError("Each environmental reading must be an object.")
 52            records.append((sensorid, reading.get("time"), name, reading.get("value")))
 53    return records

 56def clean_environment_readings(readings):
 57    """Clean extracted environmental readings and return CO2-ready data.
 58
 59    Args:
 60        readings (pandas.DataFrame): Long-form readings with ``sensorid``,
 61            ``timestamp``, ``variable`` and ``value`` columns. ``timestamp``
 62            values must be numeric Unix epoch seconds; nonnumeric timestamps
 63            are counted as invalid and excluded.
 64
 65    Returns:
 66        tuple: A wide CO2/temperature/humidity DataFrame and a dictionary of
 67            counts for invalid rows, exact duplicates, conflicts and
 68            non-positive CO2 values.
 69
 70    Raises:
 71        KeyError: If a required long-form column is absent.
 72    """
 73    required = {'sensorid', 'timestamp', 'variable', 'value'}
 74    missing = sorted(required.difference(readings.columns))
 75    if missing:
 76        raise KeyError(f'Readings are missing required columns: {missing}')
 77    frame = readings[list(required)].copy()
 78    frame['timestamp'] = pd.to_datetime(
 79        pd.to_numeric(frame['timestamp'], errors='coerce'),
 80        unit='s', utc=True, errors='coerce'
 81    )
 82    frame['value'] = pd.to_numeric(frame['value'], errors='coerce')
 83    invalid = frame[['sensorid', 'timestamp', 'variable', 'value']].isna().any(axis=1)
 84    invalid |= ~np.isfinite(frame['value'].fillna(np.nan))
 85    invalid_count = int(invalid.sum())
 86    frame = frame.loc[~invalid].copy()
 87    exact_duplicates = int(frame.duplicated().sum())
 88    frame = frame.drop_duplicates()
 89    keys = ['sensorid', 'timestamp', 'variable']
 90    conflicts = frame.duplicated(keys, keep=False)
 91    conflicting_count = int(conflicts.sum())
 92    frame = frame.loc[~conflicts]
 93    wide = frame.pivot(
 94        index=['sensorid', 'timestamp'], columns='variable', values='value'
 95    ).reset_index()
 96    co2 = wide.rename(columns={'Carbon dioxide': 'co2_ppm'}).sort_values(
 97        ['sensorid', 'timestamp']
 98    )
 99    nonpositive_count = int(co2['co2_ppm'].le(0).sum()) if 'co2_ppm' in co2 else 0
100    if 'co2_ppm' in co2:
101        co2 = co2.loc[co2['co2_ppm'].gt(0)].copy()
102    return co2, {
103        'invalid readings': invalid_count,
104        'exact duplicates removed': exact_duplicates,
105        'conflicting readings excluded': conflicting_count,
106        'nonpositive CO2 excluded': nonpositive_count,
107    }

110def build_forecast_features(sensor_series, horizon='20min', tolerance='2min'):
111    """Build backward-matched historical features and measured targets.
112
113    Args:
114        sensor_series (pandas.DataFrame): Timestamp-indexed sensor data with
115            ``co2_ppm``, ``Temperature`` and ``Humidity`` columns.
116        horizon (str or pandas.Timedelta): Desired forecast horizon.
117        tolerance (str or pandas.Timedelta): Maximum backward/target matching
118            distance; no interpolation is performed.
119
120    Returns:
121        pandas.DataFrame: Issue times, backward CO2 lags, elapsed-time rate,
122            desired target times and nearest measured targets.
123
124    Raises:
125        KeyError: If the sensor data lacks a required column.
126        ValueError: If timestamps are invalid or the sensor data is empty.
127    """
128    required = {'co2_ppm', 'Temperature', 'Humidity'}
129    missing = sorted(required.difference(sensor_series.columns))
130    if missing:
131        raise KeyError(f'Sensor data is missing required columns: {missing}')
132    if sensor_series.empty:
133        raise ValueError('Sensor data must not be empty.')
134    horizon = pd.Timedelta(horizon)
135    tolerance = pd.Timedelta(tolerance)
136    base = sensor_series[['co2_ppm', 'Temperature', 'Humidity']].reset_index()
137    time_column = base.columns[0]
138    base = base.rename(columns={time_column: 'timestamp'})
139    base['timestamp'] = pd.to_datetime(base['timestamp'], utc=True, errors='coerce')
140    if base['timestamp'].isna().any():
141        raise ValueError('Sensor data contains invalid timestamps.')
142    base = base.sort_values('timestamp')
143    features = base.rename(columns={'timestamp': 'issue_time'}).copy()
144    for minutes in [10, 20]:
145        lookup = base[['timestamp', 'co2_ppm']].rename(columns={
146            'timestamp': f'lag_time_{minutes}',
147            'co2_ppm': f'co2_lag_{minutes}min',
148        })
149        features['lookup_time'] = features['issue_time'] - pd.Timedelta(minutes=minutes)
150        features = pd.merge_asof(
151            features.sort_values('lookup_time'), lookup,
152            left_on='lookup_time', right_on=f'lag_time_{minutes}',
153            direction='backward', tolerance=tolerance,
154        )
155    features = features.drop(columns='lookup_time').sort_values('issue_time')
156    elapsed = (
157        features['issue_time'] - features['lag_time_10']
158    ).dt.total_seconds() / 60
159    features['co2_rate_ppm_per_min'] = (
160        features['co2_ppm'] - features['co2_lag_10min']
161    ) / elapsed
162    features['desired_target_time'] = features['issue_time'] + horizon
163    targets = base[['timestamp', 'co2_ppm']].rename(columns={
164        'timestamp': 'target_time', 'co2_ppm': 'target_co2_ppm'
165    })
166    return pd.merge_asof(
167        features.sort_values('desired_target_time'), targets,
168        left_on='desired_target_time', right_on='target_time',
169        direction='nearest', tolerance=tolerance,
170    )

173def forecast_metrics(observed, predicted):
174    """Calculate forecast errors using prediction minus observation.
175
176    Args:
177        observed (array-like): Finite measured CO2 concentrations in ppm.
178        predicted (array-like): Same-length finite predictions in ppm.
179
180    Returns:
181        dict: MAE_ppm, RMSE_ppm and mean_signed_error_ppm.
182
183    Raises:
184        ValueError: If inputs are empty, nonfinite or have unequal shapes.
185
186    Examples:
187        >>> forecast_metrics([410, 410], [400, 420])['MAE_ppm']
188        10.0
189    """
190    actual, estimate = np.asarray(observed, dtype=float), np.asarray(predicted, dtype=float)
191    if actual.size == 0 or actual.shape != estimate.shape or not (np.isfinite(actual).all() and np.isfinite(estimate).all()):
192        raise ValueError('Metrics require nonempty, equal-shaped finite inputs.')
193    error = estimate - actual
194    return {'MAE_ppm': float(np.abs(error).mean()),
195            'RMSE_ppm': float(np.sqrt(np.mean(error**2))),
196            'mean_signed_error_ppm': float(error.mean())}

199def add_rate_features(frame):
200    """Add 10- and 20-minute backward differences using actual times.
201
202    Args:
203        frame (pandas.DataFrame): Prepared split with issue_time, lag_time_20,
204            co2_ppm and co2_lag_20min. Times must include timezone information.
205
206    Returns:
207        pandas.DataFrame: Copy with co2_rate_20_ppm_per_min added.
208
209    Raises:
210        KeyError: If required columns are absent.
211        ValueError: If lag elapsed times are invalid or nonpositive.
212    """
213    result = frame.copy()
214    elapsed = (pd.to_datetime(result['issue_time'], utc=True) - pd.to_datetime(result['lag_time_20'], utc=True)).dt.total_seconds() / 60
215    if elapsed.isna().any() or (elapsed <= 0).any():
216        raise ValueError('Backward rates require valid historical timestamps.')
217    if {'lag_time_10', 'co2_lag_10min'}.issubset(result.columns):
218        elapsed_10 = (pd.to_datetime(result['issue_time'], utc=True) - pd.to_datetime(result['lag_time_10'], utc=True)).dt.total_seconds() / 60
219        if elapsed_10.isna().any() or (elapsed_10 <= 0).any():
220            raise ValueError('Backward rates require valid historical timestamps.')
221        result['co2_rate_ppm_per_min'] = (result.co2_ppm - result.co2_lag_10min) / elapsed_10
222    result['co2_rate_20_ppm_per_min'] = (result.co2_ppm - result.co2_lag_20min) / elapsed
223    return result

226def validate_prepared_split(frame, name='split'):
227    """Validate a prepared chronological split before model fitting.
228
229    Args:
230        frame (pandas.DataFrame): Prepared rows containing issue/target times,
231            backward lag times and finite model inputs in ppm or source units.
232        name (str): Label used in validation error messages.
233
234    Returns:
235        None: The function returns only when all structural checks pass.
236
237    Raises:
238        KeyError: If a required prepared-data column is missing.
239        ValueError: If the split is empty, contains invalid timestamps,
240            future-looking lags, invalid horizons or non-finite values.
241    """
242    required = {
243        'issue_time', 'target_time', 'target_co2_ppm', 'co2_ppm',
244        'co2_lag_10min', 'co2_lag_20min', 'lag_time_10', 'lag_time_20',
245        'Temperature', 'Humidity',
246    }
247    missing = sorted(required.difference(frame.columns))
248    if missing:
249        raise KeyError(f'{name} is missing required columns: {missing}')
250    if frame.empty:
251        raise ValueError(f'{name} must not be empty.')
252    times = frame[['issue_time', 'target_time', 'lag_time_10', 'lag_time_20']].apply(
253        pd.to_datetime, utc=True, errors='coerce'
254    )
255    if times.isna().any().any():
256        raise ValueError(f'{name} contains invalid timestamps.')
257    if (times['target_time'] <= times['issue_time']).any():
258        raise ValueError(f'{name} contains a target at or before its issue time.')
259    if (times['lag_time_10'] > times['issue_time']).any() or (times['lag_time_20'] > times['issue_time']).any():
260        raise ValueError(f'{name} contains historical measurements after its issue time.')
261    numeric = frame[['target_co2_ppm', 'co2_ppm', 'co2_lag_10min', 'co2_lag_20min', 'Temperature', 'Humidity']]
262    if not np.isfinite(numeric.to_numpy(dtype=float)).all():
263        raise ValueError(f'{name} contains missing or nonfinite numeric values.')

266def compare_models(train, validation, test, output_dir):
267    """Tune linear/tree forecasts on validation, then score the chosen model.
268
269    Args:
270        train (pandas.DataFrame): Earliest prepared chronological split.
271        validation (pandas.DataFrame): Later tuning split with measured targets.
272        test (pandas.DataFrame): Final withheld split; never used for selection.
273        output_dir (str or pathlib.Path): Destination for result CSVs and figures.
274
275    Returns:
276        dict: Fitted selected model (None for persistence), selected feature
277            names, validation experiments, final test metrics and predictions.
278
279    Raises:
280        ValueError: If splits overlap, inputs are nonfinite or metrics invalid.
281        KeyError: If a required prepared-data column is missing.
282        OSError: If output files cannot be written.
283
284    Notes:
285        Uses fixed tree depths and minimum leaf sizes. Rate features are linear
286        combinations of CO2 history (approximately so for irregular intervals),
287        so they need not improve ordinary linear regression. Decision trees
288        cannot extrapolate beyond training leaf averages. Repeated runs reuse
289        this same test period; do not tune in response to its results.
290    """
291    train, validation, test = [add_rate_features(p) for p in (train, validation, test)]
292    for label, part in [('train', train), ('validation', validation), ('test', test)]:
293        validate_prepared_split(part, label)
294    for earlier, later in [(train, validation), (validation, test)]:
295        if pd.to_datetime(earlier.target_time, utc=True).max() >= pd.to_datetime(later.issue_time, utc=True).min():
296            raise ValueError('Splits must be chronological with target-boundary purging.')
297    history = ['co2_ppm', 'co2_lag_10min', 'co2_lag_20min']
298    environment = history + ['Temperature', 'Humidity']
299    feature_sets = {'history': history, 'history_environment': environment,
300                    'history_environment_rate10': environment + ['co2_rate_ppm_per_min'],
301                    'history_environment_rate20': environment + ['co2_rate_20_ppm_per_min']}
302    experiments, candidates = [], {}
303
304    def record(name, kind, columns, model, depth=None, leaf=None):
305        """Record a candidate's training/validation errors and runtime only."""
306        started = perf_counter()
307        if model is not None:
308            model.fit(train[columns], train.target_co2_ppm)
309        fit_seconds = perf_counter() - started
310        for label, part in [('train', train), ('validation', validation)]:
311            started = perf_counter()
312            prediction = part.co2_ppm.to_numpy(copy=True) if model is None else model.predict(part[columns])
313            seconds = perf_counter() - started
314            experiments.append({'candidate': name, 'model': kind, 'features': ', '.join(columns),
315                'max_depth': depth, 'min_samples_leaf': leaf, 'split': label,
316                **forecast_metrics(part.target_co2_ppm, prediction),
317                'fit_seconds': fit_seconds, 'prediction_seconds': seconds})
318        candidates[name] = (model, columns)
319
320    record('persistence', 'persistence', ['co2_ppm'], None)
321    for feature_name, columns in feature_sets.items():
322        record('linear_' + feature_name, 'linear', columns, LinearRegression())
323        for depth in [2, 4, 6, 8, None]:
324            for leaf in [5, 20, 50]:
325                name = f'tree_{feature_name}_depth{depth}_leaf{leaf}'
326                record(name, 'tree', columns, DecisionTreeRegressor(max_depth=depth, min_samples_leaf=leaf, random_state=42), depth, leaf)
327    results = pd.DataFrame(experiments)
328    ranking = results.loc[results.split == 'validation'].sort_values(['MAE_ppm', 'candidate']).reset_index(drop=True)
329    selected_name = ranking.iloc[0]['candidate']
330    model, columns = candidates[selected_name]
331    # Freeze selection before touching final test targets. Retain the training-only fit.
332    summaries, predictions = [], pd.DataFrame({'issue_time': test.issue_time, 'target_time': test.target_time, 'observed_co2_ppm': test.target_co2_ppm})
333    for name, estimator, cols in [('persistence', None, ['co2_ppm']), (selected_name, model, columns)]:
334        if name in predictions.columns:
335            continue
336        started = perf_counter()
337        pred = test.co2_ppm.to_numpy(copy=True) if estimator is None else estimator.predict(test[cols])
338        seconds = perf_counter() - started
339        predictions[name] = pred
340        summaries.append({'candidate': name, **forecast_metrics(test.target_co2_ppm, pred), 'prediction_seconds': seconds})
341    # Compare the frozen models over calendar weeks to show whether the test
342    # improvement is consistent rather than driven by one short interval.
343    temporal_errors = pd.DataFrame({
344        'target_time': pd.to_datetime(predictions.target_time, utc=True),
345        'persistence_error_ppm': predictions.persistence - predictions.observed_co2_ppm,
346        'selected_error_ppm': predictions[selected_name] - predictions.observed_co2_ppm,
347    })
348    temporal_errors['target_week'] = temporal_errors['target_time'].dt.strftime('%Y-W%U')
349    weekly_errors = temporal_errors.groupby('target_week', sort=True).agg(
350        rows=('target_time', 'size'),
351        persistence_mae_ppm=('persistence_error_ppm', lambda values: np.abs(values).mean()),
352        selected_mae_ppm=('selected_error_ppm', lambda values: np.abs(values).mean()),
353    ).reset_index()
354    weekly_errors['improvement_ppm'] = (
355        weekly_errors['persistence_mae_ppm'] - weekly_errors['selected_mae_ppm']
356    )
357    weekly_errors['improvement_percent'] = np.where(
358        weekly_errors['persistence_mae_ppm'].ne(0),
359        100 * weekly_errors['improvement_ppm'] / weekly_errors['persistence_mae_ppm'],
360        np.nan,
361    )
362
363    # Report rising periods using an a-priori historical rate threshold, never a target-based filter.
364    rising = test.co2_rate_ppm_per_min > 1.0
365    rising_rows = []
366    if rising.any():
367        for row in summaries:
368            name = row['candidate']
369            rising_rows.append({'candidate': name, 'rows': int(rising.sum()), **forecast_metrics(test.loc[rising, 'target_co2_ppm'], predictions.loc[rising, name])})
370    # Stress only input features at validation time; this is synthetic sensitivity,
371    # not a new accuracy estimate or a basis for retuning after test evaluation.
372    noise_rows = []
373    rng = np.random.default_rng(42)
374    for sigma in [0., 5., 10.]:
375        noisy = validation.copy()
376        noisy[history] = noisy[history] + rng.normal(0, sigma, (len(noisy), len(history)))
377        elapsed = (pd.to_datetime(noisy.issue_time, utc=True) - pd.to_datetime(noisy.lag_time_10, utc=True)).dt.total_seconds() / 60
378        noisy['co2_rate_ppm_per_min'] = (noisy.co2_ppm - noisy.co2_lag_10min) / elapsed
379        noisy = add_rate_features(noisy)
380        pred = noisy.co2_ppm.to_numpy() if model is None else model.predict(noisy[columns])
381        noise_rows.append({'input_noise_std_ppm': sigma, **forecast_metrics(noisy.target_co2_ppm, pred)})
382    output = Path(output_dir)
383    output.mkdir(parents=True, exist_ok=True)
384    results.to_csv(output / 'model_experiments.csv', index=False)
385    ranking.to_csv(output / 'validation_ranking.csv', index=False)
386    pd.DataFrame(summaries).to_csv(output / 'final_test_metrics.csv', index=False)
387    predictions.to_csv(output / 'final_test_predictions.csv', index=False)
388    weekly_errors.to_csv(output / 'test_error_by_week.csv', index=False)
389    pd.DataFrame(rising_rows).to_csv(output / 'rising_co2_test_metrics.csv', index=False)
390    pd.DataFrame(noise_rows).to_csv(output / 'validation_noise_sensitivity.csv', index=False)
391    if hasattr(model, 'coef_'):
392        pd.DataFrame({'feature': columns, 'coefficient': model.coef_}).to_csv(output / 'selected_model_coefficients.csv', index=False)
393    elif hasattr(model, 'feature_importances_'):
394        pd.DataFrame({'feature': columns, 'importance': model.feature_importances_}).to_csv(output / 'selected_model_importance.csv', index=False)
395    return {'selected_name': selected_name, 'model': model, 'features': columns,
396            'experiments': results, 'ranking': ranking, 'test_metrics': pd.DataFrame(summaries),
397            'test_predictions': predictions, 'noise_sensitivity': pd.DataFrame(noise_rows),
398            'test_error_by_week': weekly_errors}

401def plot_model_results(comparison, output_dir):
402    """Save and display validation comparisons and final test diagnostic plots.
403
404    Args:
405        comparison (dict): Results returned by compare_models.
406        output_dir (str or pathlib.Path): Folder for PNG figures.
407
408    Returns:
409        None: Saves validation_comparison.png and test_diagnostics.png.
410
411    Raises:
412        ImportError: If matplotlib is not installed.
413        OSError: If figures cannot be saved.
414    """
415    import matplotlib.pyplot as plt
416    output = Path(output_dir)
417    output.mkdir(parents=True, exist_ok=True)
418    best = comparison['ranking'].groupby('model', sort=False).head(1).copy()
419    best['family_label'] = best['model'].replace({
420        'persistence': 'Persistence',
421        'linear': 'Linear',
422        'tree': 'Decision tree',
423    })
424    best_value = best['MAE_ppm'].min()
425    best['excess_ppm'] = best['MAE_ppm'] - best_value
426
427    fig, axes = plt.subplots(
428        1, 2, figsize=(12, 4.5), gridspec_kw={'width_ratios': [1.2, 1]}
429    )
430
431    # A tight y-axis and labels make small absolute differences visible.
432    bars = axes[0].bar(
433        best['family_label'], best['MAE_ppm'],
434        color=['#4c78a8', '#59a14f', '#f28e2b'],
435    )
436    value_range = max(best['MAE_ppm'].max() - best['MAE_ppm'].min(), 0.1)
437    axes[0].set_ylim(
438        best['MAE_ppm'].min() - 0.15 * value_range,
439        best['MAE_ppm'].max() + 0.25 * value_range,
440    )
441    axes[0].bar_label(bars, fmt='%.3f ppm', padding=3, fontsize=9)
442    axes[0].set_ylabel('Validation MAE (ppm)')
443    axes[0].set_title('Absolute error (zoomed scale)')
444    axes[0].grid(axis='y', alpha=.25)
445
446    # A zero-based difference view directly shows the ranking gaps.
447    bars = axes[1].bar(best['family_label'], best['excess_ppm'], color='#9c755f')
448    axes[1].bar_label(bars, fmt='%.3f ppm', padding=3, fontsize=9)
449    axes[1].set_ylim(0, max(best['excess_ppm'].max() * 1.35, 0.1))
450    axes[1].set_ylabel('Extra MAE above best family (ppm)')
451    axes[1].set_title('Difference from best configuration')
452    axes[1].grid(axis='y', alpha=.25)
453
454    for axis in axes:
455        axis.tick_params(axis='x', rotation=15)
456    fig.suptitle('Best validation configuration per model family')
457    fig.tight_layout()
458    fig.savefig(output / 'validation_comparison.png', dpi=160)
459    plt.show()
460    predictions = comparison['test_predictions']
461    name = comparison['selected_name']
462    error = predictions[name] - predictions.observed_co2_ppm
463    fig, axes = plt.subplots(3, 1, figsize=(12, 10))
464    import matplotlib.dates as mdates
465    times = pd.to_datetime(predictions.target_time, utc=True)

467    axes[0].plot(times, predictions[name], label='Selected forecast', linewidth=.8, alpha=.7)
468    axes[0].set(ylabel='CO2 (ppm)', title='Final test: measured and forecast'); axes[0].legend()
469    axes[1].plot(times, error, linewidth=.6); axes[1].axhline(0, color='black', linewidth=.8)
470    axes[1].set(ylabel='Prediction minus measured (ppm)', xlabel='Target time (UTC)')
471    for ax in axes[:2]:
472        locator = mdates.AutoDateLocator(minticks=4, maxticks=7)
473        ax.xaxis.set_major_locator(locator)
474        ax.xaxis.set_major_formatter(mdates.ConciseDateFormatter(locator))
475    axes[2].hist(error, bins=40); axes[2].set(xlabel='Residual (ppm)', ylabel='Count')
476    fig.tight_layout(); fig.savefig(output / 'test_diagnostics.png', dpi=160)
477    plt.show()

128    """
129    required = {'co2_ppm', 'Temperature', 'Humidity'}
130    missing = sorted(required.difference(sensor_series.columns))
131    if missing:
132        raise KeyError(f'Sensor data is missing required columns: {missing}')

134        raise ValueError('Sensor data must not be empty.')

175    """Calculate forecast errors using prediction minus observation.
176
177    Args:
178        observed (array-like): Finite measured CO2 concentrations in ppm.
179        predicted (array-like): Same-length finite predictions in ppm.
180
181    Returns:
182        dict: MAE_ppm, RMSE_ppm and mean_signed_error_ppm.
183
184    Raises:
185        ValueError: If inputs are empty, nonfinite or have unequal shapes.
186
187    Examples:
188        >>> forecast_metrics([410, 410], [400, 420])['MAE_ppm']

190    """
191    actual, estimate = np.asarray(observed, dtype=float), np.asarray(predicted, dtype=float)

195    return {'MAE_ppm': float(np.abs(error).mean()),
196            'RMSE_ppm': float(np.sqrt(np.mean(error**2))),
197            'mean_signed_error_ppm': float(error.mean())}

267def compare_models(train, validation, test, output_dir):
268    """Tune linear/tree forecasts on validation, then score the chosen model.
269
270    Args:
271        train (pandas.DataFrame): Earliest prepared chronological split.
272        validation (pandas.DataFrame): Later tuning split with measured targets.
273        test (pandas.DataFrame): Final withheld split; never used for selection.
274        output_dir (str or pathlib.Path): Destination for result CSVs and figures.
275
276    Returns:
277        dict: Fitted selected model (None for persistence), selected feature
278            names, validation experiments, final test metrics and predictions.
279
280    Raises:
281        ValueError: If splits overlap, inputs are nonfinite or metrics invalid.
282        KeyError: If a required prepared-data column is missing.
283        OSError: If output files cannot be written.
284
285    Notes:
286        Uses fixed tree depths and minimum leaf sizes. Rate features are linear
287        combinations of CO2 history (approximately so for irregular intervals),
288        so they need not improve ordinary linear regression. Decision trees
289        cannot extrapolate beyond training leaf averages. Repeated runs reuse
290        this same test period; do not tune in response to its results.
291    """
292    train, validation, test = [add_rate_features(p) for p in (train, validation, test)]
293    for label, part in [('train', train), ('validation', validation), ('test', test)]:
294        validate_prepared_split(part, label)
295    for earlier, later in [(train, validation), (validation, test)]:
296        if pd.to_datetime(earlier.target_time, utc=True).max() >= pd.to_datetime(later.issue_time, utc=True).min():
297            raise ValueError('Splits must be chronological with target-boundary purging.')
298    history = ['co2_ppm', 'co2_lag_10min', 'co2_lag_20min']
299    environment = history + ['Temperature', 'Humidity']
300    feature_sets = {'history': history, 'history_environment': environment,
301                    'history_environment_rate10': environment + ['co2_rate_ppm_per_min'],
302                    'history_environment_rate20': environment + ['co2_rate_20_ppm_per_min']}
303    experiments, candidates = [], {}
304
305    def record(name, kind, columns, model, depth=None, leaf=None):
306        """Record a candidate's training/validation errors and runtime only."""
307        started = perf_counter()
308        if model is not None:
309            model.fit(train[columns], train.target_co2_ppm)
310        fit_seconds = perf_counter() - started
311        for label, part in [('train', train), ('validation', validation)]:
312            started = perf_counter()
313            prediction = part.co2_ppm.to_numpy(copy=True) if model is None else model.predict(part[columns])
314            seconds = perf_counter() - started
315            experiments.append({'candidate': name, 'model': kind, 'features': ', '.join(columns),
316                'max_depth': depth, 'min_samples_leaf': leaf, 'split': label,
317                **forecast_metrics(part.target_co2_ppm, prediction),
318                'fit_seconds': fit_seconds, 'prediction_seconds': seconds})
319        candidates[name] = (model, columns)
320
321    record('persistence', 'persistence', ['co2_ppm'], None)
322    for feature_name, columns in feature_sets.items():
323        record('linear_' + feature_name, 'linear', columns, LinearRegression())
324        for depth in [2, 4, 6, 8, None]:
325            for leaf in [5, 20, 50]:
326                name = f'tree_{feature_name}_depth{depth}_leaf{leaf}'
327                record(name, 'tree', columns, DecisionTreeRegressor(max_depth=depth, min_samples_leaf=leaf, random_state=42), depth, leaf)
328    results = pd.DataFrame(experiments)
329    ranking = results.loc[results.split == 'validation'].sort_values(['MAE_ppm', 'candidate']).reset_index(drop=True)
330    selected_name = ranking.iloc[0]['candidate']
331    model, columns = candidates[selected_name]
332    # Freeze selection before touching final test targets. Retain the training-only fit.
333    summaries, predictions = [], pd.DataFrame({'issue_time': test.issue_time, 'target_time': test.target_time, 'observed_co2_ppm': test.target_co2_ppm})
334    for name, estimator, cols in [('persistence', None, ['co2_ppm']), (selected_name, model, columns)]:
335        if name in predictions.columns:
336            continue
337        started = perf_counter()
338        pred = test.co2_ppm.to_numpy(copy=True) if estimator is None else estimator.predict(test[cols])
339        seconds = perf_counter() - started
340        predictions[name] = pred
341        summaries.append({'candidate': name, **forecast_metrics(test.target_co2_ppm, pred), 'prediction_seconds': seconds})
342    # Compare the frozen models over calendar weeks to show whether the test
343    # improvement is consistent rather than driven by one short interval.
344    temporal_errors = pd.DataFrame({
345        'target_time': pd.to_datetime(predictions.target_time, utc=True),
346        'persistence_error_ppm': predictions.persistence - predictions.observed_co2_ppm,
347        'selected_error_ppm': predictions[selected_name] - predictions.observed_co2_ppm,
348    })
349    temporal_errors['target_week'] = temporal_errors['target_time'].dt.strftime('%Y-W%U')
350    weekly_errors = temporal_errors.groupby('target_week', sort=True).agg(
351        rows=('target_time', 'size'),
352        persistence_mae_ppm=('persistence_error_ppm', lambda values: np.abs(values).mean()),
353        selected_mae_ppm=('selected_error_ppm', lambda values: np.abs(values).mean()),
354    ).reset_index()
355    weekly_errors['improvement_ppm'] = (
356        weekly_errors['persistence_mae_ppm'] - weekly_errors['selected_mae_ppm']
357    )
358    weekly_errors['improvement_percent'] = np.where(
359        weekly_errors['persistence_mae_ppm'].ne(0),
360        100 * weekly_errors['improvement_ppm'] / weekly_errors['persistence_mae_ppm'],
361        np.nan,
362    )
363
364    # Report rising periods using an a-priori historical rate threshold, never a target-based filter.
365    rising = test.co2_rate_ppm_per_min > 1.0
366    rising_rows = []
367    if rising.any():
368        for row in summaries:
369            name = row['candidate']
370            rising_rows.append({'candidate': name, 'rows': int(rising.sum()), **forecast_metrics(test.loc[rising, 'target_co2_ppm'], predictions.loc[rising, name])})
371    # Stress only input features at validation time; this is synthetic sensitivity,
372    # not a new accuracy estimate or a basis for retuning after test evaluation.
373    noise_rows = []
374    rng = np.random.default_rng(42)
375    for sigma in [0., 5., 10.]:
376        noisy = validation.copy()
377        noisy[history] = noisy[history] + rng.normal(0, sigma, (len(noisy), len(history)))
378        elapsed = (pd.to_datetime(noisy.issue_time, utc=True) - pd.to_datetime(noisy.lag_time_10, utc=True)).dt.total_seconds() / 60
379        noisy['co2_rate_ppm_per_min'] = (noisy.co2_ppm - noisy.co2_lag_10min) / elapsed
380        noisy = add_rate_features(noisy)
381        pred = noisy.co2_ppm.to_numpy() if model is None else model.predict(noisy[columns])
382        noise_rows.append({'input_noise_std_ppm': sigma, **forecast_metrics(noisy.target_co2_ppm, pred)})
383    output = Path(output_dir)
384    output.mkdir(parents=True, exist_ok=True)
385    results.to_csv(output / 'model_experiments.csv', index=False)
386    ranking.to_csv(output / 'validation_ranking.csv', index=False)
387    pd.DataFrame(summaries).to_csv(output / 'final_test_metrics.csv', index=False)
388    predictions.to_csv(output / 'final_test_predictions.csv', index=False)
389    weekly_errors.to_csv(output / 'test_error_by_week.csv', index=False)
390    pd.DataFrame(rising_rows).to_csv(output / 'rising_co2_test_metrics.csv', index=False)
391    pd.DataFrame(noise_rows).to_csv(output / 'validation_noise_sensitivity.csv', index=False)
392    if hasattr(model, 'coef_'):
393        pd.DataFrame({'feature': columns, 'coefficient': model.coef_}).to_csv(output / 'selected_model_coefficients.csv', index=False)
394    elif hasattr(model, 'feature_importances_'):
395        pd.DataFrame({'feature': columns, 'importance': model.feature_importances_}).to_csv(output / 'selected_model_importance.csv', index=False)
396    return {'selected_name': selected_name, 'model': model, 'features': columns,
397            'experiments': results, 'ranking': ranking, 'test_metrics': pd.DataFrame(summaries),
398            'test_predictions': predictions, 'noise_sensitivity': pd.DataFrame(noise_rows),
399            'test_error_by_week': weekly_errors}
